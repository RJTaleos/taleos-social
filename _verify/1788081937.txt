"""Read-only health check for the Taleos publisher.

Confirms the Page token, the Instagram link and the GitHub image host all work.
Publishes NOTHING. Safe to run any time. Never prints the tokens.
"""
import os
import sys
import time

import requests

ENV_PATH = os.path.expanduser('~/.config/taleos/meta.env')
GRAPH = 'https://graph.facebook.com/v21.0'

def load_env(path=ENV_PATH):
    if not os.path.exists(path):
        sys.exit(f'✗ No config file at {path} — do step 5 of GUIDE.md first.')
    env = {}
    for line in open(path):
        line = line.strip()
        if line and not line.startswith('#') and '=' in line:
            k, v = line.split('=', 1)
            env[k.strip()] = v.strip()
    env.setdefault('GITHUB_BRANCH', 'main')
    missing = [k for k in ('META_PAGE_ID', 'META_IG_USER_ID', 'META_PAGE_TOKEN',
                           'GITHUB_TOKEN', 'GITHUB_REPO') if not env.get(k)]
    if missing:
        sys.exit(f'✗ {path} is missing: {", ".join(missing)} — see GUIDE.md step 5.')
    return env

def main():
    env = load_env()
    ok = True

    # 1 — Facebook Page
    r = requests.get(f'{GRAPH}/{env["META_PAGE_ID"]}',
                     params={'fields': 'name,id',
                             'access_token': env['META_PAGE_TOKEN']}, timeout=30)
    if r.ok:
        d = r.json()
        print(f'✓ Facebook Page ....... {d.get("name")} (id {d.get("id")})')
    else:
        ok = False
        print(f'✗ Facebook Page ....... {r.json().get("error", {}).get("message", r.text)[:140]}')

    # 2 — Instagram business account
    r = requests.get(f'{GRAPH}/{env["META_IG_USER_ID"]}',
                     params={'fields': 'username,id',
                             'access_token': env['META_PAGE_TOKEN']}, timeout=30)
    if r.ok:
        d = r.json()
        print(f'✓ Instagram ........... @{d.get("username")} (id {d.get("id")})')
    else:
        ok = False
        print(f'✗ Instagram ........... {r.json().get("error", {}).get("message", r.text)[:140]}')

    # 3 — GitHub host: upload a tiny test file, fetch it publicly, delete it
    try:
        import gh_host
        key = f'_verify/{int(time.time())}.txt'
        url = gh_host.upload(env, __file__, key)  # any small file works; reuse this one
        pub = requests.get(url, timeout=30)
        gh_host.delete(env, key)
        if pub.ok and b'health check' in pub.content:
            ctype = pub.headers.get('Content-Type', '?')
            print(f'✓ GitHub hosting ...... {env["GITHUB_REPO"]} — upload + public fetch OK '
                  f'(served as {ctype.split(";")[0]})')
        else:
            ok = False
            print(f'✗ GitHub hosting ...... uploaded, but public fetch failed '
                  f'(HTTP {pub.status_code}) — is the repo Public?')
    except Exception as e:
        ok = False
        print(f'✗ GitHub hosting ...... {type(e).__name__}: {str(e)[:160]}')

    print('\nALL GOOD — ready to publish.' if ok
          else '\nSomething failed — see the Troubleshooting table in GUIDE.md, '
               'or paste this output to Peach (it contains no secrets).')
    sys.exit(0 if ok else 1)

if __name__ == '__main__':
    main()
